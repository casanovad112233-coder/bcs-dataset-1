## 6. PostgreSQL on Supabase — Recommended Schema (for migration)

`instructions/PROMPT_FOR_CLAUDE_CODEX_SUPABASE.md:210` — explicit SQL for copy-paste.

```sql
-- enable extensions
create extension if not exists pgcrypto;

-- 1. subjects (10 rows, ground truth)
create table subjects (
  id smallint primary key check (id between 1 and 10),
  subject_en text not null,
  subject_bn text not null
);

-- 2. exams
create table exams (
  id uuid primary key default gen_random_uuid(),
  slug text unique not null check (slug ~ '^\d+(st|nd|rd|th)_bcs$'), -- e.g. 50th_bcs
  title text not null, -- "50th BCS"
  date date,
  total_marks smallint,
  set_code text, -- Bengali
  total_questions smallint not null,
  created_at timestamptz default now()
);
create index on exams (slug);

-- 3. questions (5,350 rows, grows to 10k+)
create table questions (
  id uuid primary key default gen_random_uuid(),
  exam_id uuid not null references exams(id) on delete cascade,
  exam_slug text not null, -- denormalized for filtering, equals exams.slug
  question_number smallint not null,
  subject_id smallint not null references subjects(id),
  question text not null, -- empty only if has_question_image
  option_a text not null,
  option_b text not null,
  option_c text not null,
  option_d text not null,
  correct_answer char(1) check (correct_answer in ('A','B','C','D') or correct_answer is null), -- blank defective → null
  solve_note text not null default '',
  has_question_image boolean generated always as (cardinality(question_image_paths) > 0) stored,
  has_solve_note_image boolean generated always as (cardinality(solve_note_image_paths) > 0) stored,
  has_image boolean generated always as (cardinality(question_image_paths) > 0 or cardinality(solve_note_image_paths) > 0) stored,
  question_image_paths text[] not null default '{}',
  solve_note_image_paths text[] not null default '{}',
  subject_en text not null, -- denormalized
  subject_bn text not null,
  search_vector tsvector, -- for Bangla+English FTS, see below
  unique (exam_id, question_number)
);
create index on questions (exam_id, question_number);
create index on questions (subject_id);
create index on questions (exam_slug);
create index on questions (has_image);
create index on questions (correct_answer);
create index on questions using gin (question_image_paths);
create index on questions using gin (solve_note_image_paths);
create index on questions using gin (search_vector);

-- FTS: simple config for Bangla+English (or pg_trgm for fuzzy)
-- update trigger to set search_vector = to_tsvector('simple', question || ' ' || option_a || ' ' || ... || solve_note)

-- 4. question_images (optional normalized alternative to text[])
-- If you keep text[], skip this. If you normalize:
create type image_type as enum ('question','solve');
create table question_images (
  id uuid primary key default gen_random_uuid(),
  question_id uuid not null references questions(id) on delete cascade,
  image_type image_type not null,
  storage_path text not null, -- bucket/key e.g. bcs-images/50th_bcs/q14_img1.png
  public_url text not null,
  width int, height int,
  created_at timestamptz default now()
);
create index on question_images (question_id);
```

## 9. Constraints & Notes

* Keep `correct_answer` blank/null semantics (8 defective rows). Do not coalesce.
* Images are **paths/URLs**, never base64 in DB. `images/` is source for `storage` upload.
* `exam_slug` is denormalized on `questions` for fast `WHERE exam_slug='17th_bcs'` without join.
* Assume **read-heavy** (filter by exam/subject, search, paginate `?limit=50&offset=0`). Admin writes only.
* Regeneration: fix HTML → `python -m pipeline <html>` → JSON rebuilds; or direct JSON edit for text fixes (do not re-run without re-applying manual patches noted in `instructions/memory.md`).
* See `pipeline/validate.py:23` and `.agent/AGENTS.md:100` for hard checks before any DB import.
